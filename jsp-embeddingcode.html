Embedding Code

Example.java
<%!
  String[] names = {"Green", "White", "Black", "Red"};
%>
<HTML>
  <HEAD><TITLE>Embedding Code</TITLE></HEAD>
  <BODY>
    <H1>List of people</H1>
    <TABLE BORDER="1">
      <TH>Name</TH>
      <% for (int i=0; i<names.length; i++) { %>
        <TR><TD><%= names[i]%></TD></TR>
      <% } %>
    </TABLE>
  </BODY>
</HTML>
