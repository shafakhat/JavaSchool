   * @param array2  the second arrray (<code>null</code> permitted).
   *
   * @return A boolean.
   */ public static boolean equal(final float[][] array1,
                              final float[][] array2) {
      if (array1 == null) {
          return (array2 == null);
      }
      if (array2 == null) {
          return false;
      }
      if (array1.length != array2.length) {
          return false;
      }
      for (int i = 0; i < array1.length; i++) {
          if (!Arrays.equals(array1[i], array2[i])) {
              return false;
          }
      }
      return true;
  }
}
