import java.util.Date;
public class Main {
  public static void main(String[] args) {
    long timeInMilliseconds = new Date().getTime();
    System.out.println("Time in milliseconds is : " + timeInMilliseconds);
  }
}
//Time in milliseconds is : 1235149879683
