class AutoBox5 {
  public static void main(String args[]) {
    Boolean b = true;
    if (b)
      System.out.println("b is true");
    Character ch = 'x'; // box a char
    char ch2 = ch; // unbox a char
    System.out.println("ch2 is " + ch2);
  }
}
