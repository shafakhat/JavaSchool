Example.java
double x1 = corner1.getX() - top.getX();
double x2 = corner2.getX() - top.getX();
double y1 = corner1.getY() - top.getY();
double y2 = corner2.getY() - top.getY();
returnMath.acos((x1 * x2 + y1 * y2) / Math.sqrt((x1 * x1 + y1 * y1) * (x2 * x2 + y2 * y2)));
