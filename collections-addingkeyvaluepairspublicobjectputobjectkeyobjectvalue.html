Adding Key-Value Pairs

Example.java
import java.util.HashMap;
import java.util.Map;
public class MainClass {
  public static void main(String[] a) {
    Map map = new HashMap();
    map.put("key1", "value1");
    map.put("key2", "value2");
    map.put("key3", "value3");
    System.out.println(map);
  }
}
