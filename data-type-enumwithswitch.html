enum with switch

Example.java
public class SizeSwitch {
    public static void main(String[] args) {
        Size size = Size.XL;
        switch(size){
            case S:
                System.out.println("S");
                break;
            case M:
                System.out.println("M");
                break;
            case L:
                System.out.println("L");
                break;
            case XL:
                System.out.println("XL");
                break;
            case XXL:
                System.out.println("XXL");
                break;
            case XXXL:
                System.out.println("XXXL");
                break;
        }
    }
}
enum Size {
  S, M, L, XL, XXL, XXXL;
}
