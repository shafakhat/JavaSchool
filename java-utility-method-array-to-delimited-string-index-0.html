Example.java
StringBuffer sb = newStringBuffer();
for (int i = array.length - 1; i >= 0; i--) {
    if (array[i] == 0) {
        continue;
    sb.append(array[i]);
    sb.append(i == 0 ? "" : ",");
return sb.toString();
