Example.java
import java.util.Vector;
public class MainClass {
  public static void main(String args[]) {
    Vector v = new Vector(5);
    for (int i = 0; i < 10; i++) {
      v.insertElementAt(i,0);
    }
    System.out.println(v);
    Vector v2 = new Vector();
    v2.addAll(v);
    v2.addAll(v);
    System.out.println(v2);
  }
}
