   privateString[] maxArray() {
      String[] strings = newString[MAX];
      Boolean result = Boolean.TRUE;
      for (int i = 0; i < MAX; i++) {
         strings[i] = getString(result, i);

      }
      return strings;

      returnString.valueOf(result) + i + String.valueOf(!result);
