Arrays of Strings

Example.java
public class MainClass {
  public static void main(String[] arg) {
    String[] colors = {"red", "orange", "yellow", "green", "blue", "indigo", "violet"};
    for(String s: colors){
      System.out.println(s);
    }
  }
}
