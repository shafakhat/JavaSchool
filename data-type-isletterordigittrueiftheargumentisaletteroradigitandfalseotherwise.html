isLetterOrDigit()

Example.java
public class MainClass {
  public static void main(String[] args) {
    char symbol = 'A';
    if (Character.isLetterOrDigit(symbol)) {
      System.out.println("true");
    }else{
      System.out.println("false");
    }
  }
}
